export const AMBIGUOUS_MARGIN = 0.15;

export interface Prediction {
  segment: Segment;
  probabilities: Record<Segment, number>;
  margin: number;
  ambiguous: boolean;
  imputed: FeatureKey[];
}

export function predictSegment(input: FeatureInput, model: ModelParams): Prediction {
  const imputed: FeatureKey[] = [];
  const x = model.features.map((key, i) => {
    const raw = input[key];
    if (raw === null || raw === undefined || (typeof raw === 'number' && Number.isNaN(raw))) {
      imputed.push(key);
      return model.median[i];
    }
    if (typeof raw !== 'number' || !Number.isFinite(raw)) {
      throw new Error(`Nilai ${key} bukan angka yang valid`);
    }
    return raw;
  });

  const probs = model.classes.map(() => 0);
  for (const tree of model.forest) {
    let n = 0;
    while (tree.l[n] !== -1) n = x[tree.f[n]] <= tree.t[n] ? tree.l[n] : tree.r[n];
    tree.v[n].forEach((pv, k) => {
      probs[k] += pv / model.forest.length;
    });
  }

  const probabilities = {} as Record<Segment, number>;
  model.classes.forEach((cls, k) => {
    probabilities[cls] = probs[k];
  });

  let best = 0;
  probs.forEach((p, k) => {
    if (p > probs[best]) best = k;
  });
  const sorted = [...probs].sort((a, b) => b - a);
  const margin = sorted[0] - sorted[1];

  return {
    segment: model.classes[best],
    probabilities,
    margin,
    ambiguous: margin < AMBIGUOUS_MARGIN,
    imputed,
  };

    alts.map((a) => a.values),
    weights,
    isCost,
  );
  const ranking: RankedItem[] = alts.map((a, i) => ({

    isInput: a.isInput,
    name: a.name,
    label: a.label,
    closeness: res.closeness[i],
    rank: res.rank[i],
    dPlus: res.dPlus[i],
    dMinus: res.dMinus[i],
    weighted: res.weighted[i],
  }));

    (a, b) =>
      b.closeness - a.closeness || Number(a.isInput) - Number(b.isInput) || (a.id ?? 0) - (b.id ?? 0),

function resolveWeights(segment: Segment, custom?: number[]): number[] {

    throw new InputError([`Bobot harus berjumlah ${ahp.criteria.length}`]);

}

export function normalizeWeights(weights: number[]): number[] {
  if (weights.some((w) => !Number.isFinite(w) || w < 0)) {
    throw new Error('Bobot harus angka tidak negatif');
  }
  const total = weights.reduce((s, w) => s + w, 0);
  if (total <= 0) throw new Error('Total bobot harus lebih dari nol');
  return weights.map((w) => w / total);
}

export function topsis(matrix: number[][], weights: number[], isCost: boolean[]): TopsisResult {
  const m = matrix.length;
  const n = weights.length;
  if (m === 0) throw new Error('Alternatif kosong');
  if (matrix.some((row) => row.length !== n) || isCost.length !== n) {
    throw new Error('Dimensi matriks, bobot, dan jenis kriteria tidak sama');
  }

  const norms = Array.from({ length: n }, (_, j) =>
    Math.sqrt(matrix.reduce((s, row) => s + row[j] * row[j], 0)),
  );
  const weighted = matrix.map((row) => row.map((v, j) => (norms[j] === 0 ? 0 : v / norms[j]) * weights[j]));

  const idealPositive = Array.from({ length: n }, (_, j) => {
    const col = weighted.map((r) => r[j]);
    return isCost[j] ? Math.min(...col) : Math.max(...col);
  });
  const idealNegative = Array.from({ length: n }, (_, j) => {
    const col = weighted.map((r) => r[j]);
    return isCost[j] ? Math.max(...col) : Math.min(...col);
  });

  const dist = (row: number[], ideal: number[]) =>
    Math.sqrt(row.reduce((s, v, j) => s + (v - ideal[j]) ** 2, 0));
  const dPlus = weighted.map((r) => dist(r, idealPositive));
  const dMinus = weighted.map((r) => dist(r, idealNegative));
  const closeness = dPlus.map((dp, i) => (dp + dMinus[i] === 0 ? 0 : dMinus[i] / (dp + dMinus[i])));
  const rank = closeness.map((v) => 1 + closeness.filter((o) => o > v).length);

  return { weighted, idealPositive, idealNegative, dPlus, dMinus, closeness, rank };
}
